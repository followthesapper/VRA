\subsection{Boundary Determination}

Using linear interpolation between data points:

\textbf{HIGH/TRANSITION boundary ($R^2 = 0.90$):}
\begin{align}
\rho_1 &= \rho(r=126) + \frac{\rho(r=168) - \rho(r=126)}{R^2(168) - R^2(126)} \cdot (0.90 - R^2(126)) \\
&= 0.125 + \frac{0.167 - 0.125}{0.977 - 0.821} \cdot (0.90 - 0.821) \\
&= 0.146
\end{align}

\textbf{TRANSITION/LOW boundary ($R^2 = 0.98$):}
\begin{align}
\rho_2 &= \rho(r=168) + \frac{\rho(r=504) - \rho(r=168)}{R^2(504) - R^2(168)} \cdot (0.98 - R^2(168)) \\
&= 0.167 + \frac{0.500 - 0.167}{0.988 - 0.977} \cdot (0.98 - 0.977) \\
&= 0.263
\end{align}

\subsection{Regime Trichotomy}

\begin{theorem}[Regime Map]
\label{thm:regime_map}
VRA exhibits three operational regimes characterized by $\rho = r/N$:

\begin{enumerate}
    \item \textbf{HIGH SNR} ($\rho < 0.146$):
    \begin{itemize}
        \item Requires phase-aligned bases $\mathcal{P}_a$
        \item $\sqrt{M}$ fit: $R^2 \in [0.50, 0.90]$
        \item Strong coherence, sparse frequency structure
        \item Random bases may fail (destructive interference)
    \end{itemize}

    \item \textbf{TRANSITION} ($0.146 \leq \rho < 0.263$):
    \begin{itemize}
        \item Any same-order bases work
        \item $\sqrt{M}$ fit: $R^2 \in [0.90, 0.98]$
        \item Base invariance: CV $< 10^{-15}$
        \item Flexible, forgiving to base selection
    \end{itemize}

    \item \textbf{LOW SNR} ($\rho \geq 0.263$):
    \begin{itemize}
        \item Any same-order bases work
        \item $\sqrt{M}$ fit: $R^2 \geq 0.98$
        \item Robust, excellent fit quality
        \item Dense frequency structure
    \end{itemize}
\end{enumerate}
\end{theorem}

\subsection{Cross-Validation Metrics}

The boundaries are validated by three independent metrics:

\textbf{1. Base variance:}
\begin{itemize}
    \item HIGH SNR ($r=8$): CV not applicable (phase alignment required)
    \item TRANSITION/LOW ($r \geq 126$): CV $\approx 0\%$
\end{itemize}

\textbf{2. Precision/recall:}
\begin{itemize}
    \item All regimes: Precision = 100\% (0 false positives)
    \item Recall $\geq 98\%$ across all test points
\end{itemize}

\textbf{3. $\sqrt{M}$ fit quality:}
\begin{itemize}
    \item Monotonic increase in $R^2$ as $\rho$ increases
    \item Sharp transition at empirical boundaries
\end{itemize}

\subsection{Uncertainty Analysis}

\textbf{Boundary confidence:} With only 4 data points, boundaries carry $\pm 5\%$ uncertainty. Additional tests at $r \in \{100, 150, 200, 250\}$ would tighten estimates.

\textbf{Single-modulus limitation:} All tests used $N=1009$. Cross-modulus validation ($N \in \{997, 1013, 2017\}$) would establish generalization.

\subsection{Practitioner Decision Tree}

\begin{algorithm}[h]
\caption{VRA Regime Selection}
\begin{algorithmic}
\STATE \textbf{Input:} Modulus $N$, order $r$
\STATE Compute $\rho \gets r / N$
\IF{$\rho < 0.15$}
    \STATE Regime $\gets$ HIGH\_SNR
    \STATE Bases $\gets$ phase\_aligned($a, r$)
    \STATE $L_{\max} \gets 8192$
\ELSIF{$0.15 \leq \rho < 0.26$}
    \STATE Regime $\gets$ TRANSITION
    \STATE Bases $\gets$ any\_same\_order($N, r, M$)
    \STATE $L_{\max} \gets 262144$
\ELSE
    \STATE Regime $\gets$ LOW\_SNR
    \STATE Bases $\gets$ any\_same\_order($N, r, M$)
    \STATE $L_{\max} \gets 262144+$
\ENDIF
\STATE $R \gets \lfloor 0.5 \log_2(L) \rfloor$
\STATE Budget $M$ for target SNR
\RETURN Regime, Bases, $L$, $R$
\end{algorithmic}
\end{algorithm}
