We validate VRA through 108 experiments spanning 4 regime points, 3 FFT lengths, 3 window functions, and multiple $M$ values.

\subsection{Experimental Design}

\textbf{Test matrix:}
\begin{itemize}
    \item \textbf{Modulus:} $N = 1009$ (prime)
    \item \textbf{Orders:} $r \in \{8, 126, 168, 504\}$ covering HIGH/TRANSITION/LOW regimes
    \item \textbf{$M$ values:} $\{1, 4, 8, 16, 32, 48\}$ (subset per test)
    \item \textbf{FFT lengths:} $L \in \{65536, 131072, 262144\}$
    \item \textbf{Windows:} Hann, Hamming, Blackman
    \item \textbf{Seed:} $x_0 = 1$
    \item \textbf{Zero-padding:} Factor of 8
\end{itemize}

\textbf{Total configurations:} $4 \times 6 \times 3 \times 3 = 216$ potential tests; 108 executed (targeted subset).

\subsection{Metrics Collected}

For each configuration:
\begin{enumerate}
    \item Concentration $C_M$ at each $M$
    \item $\sqrt{M}$ linear fit: slope $\alpha$, $R^2$
    \item Precision and recall at validated radius $R$
    \item Base variance (CV) across random base subsets
    \item Peak detection: TP, FP, FN counts
\end{enumerate}

\subsection{Computational Performance}

\textbf{Runtime scaling:}
\begin{itemize}
    \item Single-base FFT: $O(L \log L) \approx 0.02$s for $L=65536$
    \item $M$-base average: Linear in $M$
    \item Total: $O(ML \log L) \approx 0.64$s for $M=32$, $L=65536$
\end{itemize}

\textbf{Memory:} $O(L)$ per base; can parallelize across $M$ bases.
